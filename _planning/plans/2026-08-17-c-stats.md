```python
#!/usr/bin/env python3
"""Generate assets/data/stats.json from the ESPN exports and the Sleeper API.

Run by hand from the repo root when you want to refresh:

    python3 tools/build_stats.py --refresh   # hit the network, update the cache
    python3 tools/build_stats.py             # rebuild from the cached responses

The script aborts without writing if any self-check fails. See
_planning/specs/2026-08-17-logo-theme-seasons-stats-design.md §6.
"""

import json
import os

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
OVERRIDES_PATH = os.path.join(ROOT, "assets/data/overrides.json")
CACHE_PATH = os.path.join(ROOT, "tools/fixtures/sleeper_raw.json")
OUTPUT_PATH = os.path.join(ROOT, "assets/data/stats.json")
ESPN_PATHS = {
    "2018": os.path.join(ROOT, "assets/data/2018.json"),
    "2019": os.path.join(ROOT, "assets/data/2019.json"),
}

# Title path only. Consolation counts for nothing: spec §6.4.1.
ESPN_TITLE_TIER = "WINNERS_BRACKET"
SLEEPER_TITLE_PLACEMENTS = (None, 1)


def load_overrides(path=OVERRIDES_PATH):
    with open(path, encoding="utf-8") as handle:
        return json.load(handle)


def resolve_identities(overrides):
    """Invert the manager map into lookup tables, one per era."""
    sleeper = {}
    espn = {}
    for nickname, entry in overrides["managers"].items():
        for user_id in entry.get("sleeper", []):
            sleeper[user_id] = nickname
        for name in entry.get("espn", []):
            espn[name] = nickname
    return {"sleeper": sleeper, "espn": espn}
```

```python
def _espn_manager_name(user):
    meta = user.get("metadata") or {}
    parts = [meta.get("first_name"), meta.get("last_name")]
    return " ".join(p for p in parts if p).strip()


def espn_era(raw_by_year, ids):
    """Aggregate the two ESPN exports into per-manager totals."""
    out = {}

    def bucket(name):
        if name not in out:
            out[name] = {
                "wins": 0, "losses": 0,
                "playoffWins": 0, "playoffLosses": 0,
                "playoffYears": set(), "seasons": set(),
            }
        return out[name]

    for year, raw in sorted(raw_by_year.items()):
        season = raw["seasons"][year]
        users = {u["user_id"]: u for u in season["users"]}

        roster_owner = {}
        for roster in season["rosters"]:
            espn_name = _espn_manager_name(users.get(roster["owner_id"], {}))
            nickname = ids["espn"].get(espn_name)
            if not nickname:
                continue  # a manager who never joined the current league
            roster_owner[roster["roster_id"]] = nickname

            entry = bucket(nickname)
            entry["seasons"].add(int(year))
            settings = roster.get("settings") or {}
            # Regular season only. Never read standings[].record: spec §6.4.1.
            entry["wins"] += settings.get("wins", 0)
            entry["losses"] += settings.get("losses", 0)

        for week_entries in season["matchups_by_week"].values():
            for entry in week_entries:
                meta = entry.get("metadata") or {}
                if meta.get("playoff_tier_type") != ESPN_TITLE_TIER:
                    continue  # consolation ladders count for nothing
                nickname = roster_owner.get(entry["roster_id"])
                if not nickname:
                    continue
                bucket(nickname)["playoffYears"].add(int(year))
                if meta.get("opponent_roster_id") is None:
                    continue  # a bye is not a game and is not a loss
                if meta.get("won_matchup") == "true":
                    out[nickname]["playoffWins"] += 1
                else:
                    out[nickname]["playoffLosses"] += 1

    for entry in out.values():
        entry["playoffYears"] = sorted(entry["playoffYears"])
        entry["seasons"] = sorted(entry["seasons"])
    return out
```

```python
import urllib.request

API = "https://api.sleeper.app/v1"


def _get(path):
    with urllib.request.urlopen(f"{API}{path}", timeout=30) as resp:
        return json.load(resp)


def fetch_chain(league_id):
    """Walk previous_league_id back to the start, skipping incomplete seasons.

    The configured league is the upcoming (pre_draft) season, so it is skipped.
    ~25 GETs total against a documented ~1000/min limit. Spec §6.5.
    """
    chain = []
    current = league_id
    while current:
        league = _get(f"/league/{current}")
        if league.get("status") == "complete":
            chain.append({
                "season": league["season"],
                "leagueId": league["league_id"],
                "name": league.get("name", ""),
                "users": _get(f"/league/{current}/users"),
                "rosters": _get(f"/league/{current}/rosters"),
                "winnersBracket": _get(f"/league/{current}/winners_bracket"),
            })
        current = league.get("previous_league_id")
    return chain


def save_cache(chain, path=CACHE_PATH):
    os.makedirs(os.path.dirname(path), exist_ok=True)
    with open(path, "w", encoding="utf-8") as handle:
        json.dump(chain, handle, indent=1, sort_keys=True)


def load_cache(path=CACHE_PATH):
    with open(path, encoding="utf-8") as handle:
        return json.load(handle)
```

```python
def _title_path(bracket):
    """Only the championship path. p==3 / p==5 are placement games."""
    return [m for m in bracket if m.get("p") in SLEEPER_TITLE_PLACEMENTS]


def sleeper_era(chain, ids, roster_credits):
    out = {}
    champions = {}

    credits_by_season = {}
    for credit in roster_credits:
        credits_by_season.setdefault(credit["season"], []).append(credit)

    def bucket(name):
        if name not in out:
            out[name] = {
                "wins": 0, "losses": 0,
                "playoffWins": 0, "playoffLosses": 0,
                "playoffYears": set(), "seasons": set(),
            }
        return out[name]

    for entry in sorted(chain, key=lambda c: c["season"]):
        season = entry["season"]
        owner_of = {}

        for roster in entry["rosters"]:
            nickname = ids["sleeper"].get(roster.get("owner_id"))
            if not nickname:
                continue
            owner_of[roster["roster_id"]] = nickname

        # Extra credit for a co-owned roster: both managers get the season.
        extra = {}
        for credit in credits_by_season.get(season, []):
            extra.setdefault(credit["rosterId"], []).append(credit["creditTo"])

        for roster in entry["rosters"]:
            settings = roster.get("settings") or {}
            names = []
            if roster["roster_id"] in owner_of:
                names.append(owner_of[roster["roster_id"]])
            names.extend(extra.get(roster["roster_id"], []))
            for name in names:
                record = bucket(name)
                record["seasons"].add(int(season))
                record["wins"] += settings.get("wins", 0)
                record["losses"] += settings.get("losses", 0)

        bracket = _title_path(entry["winnersBracket"])
        for match in bracket:
            for slot in ("t1", "t2"):
                team = match.get(slot)
                if isinstance(team, int) and team in owner_of:
                    bucket(owner_of[team])["playoffYears"].add(int(season))
                    for name in extra.get(team, []):
                        bucket(name)["playoffYears"].add(int(season))
            # A bye has no w/l recorded, so nothing is counted for it.
            winner, loser = match.get("w"), match.get("l")
            if winner in owner_of:
                out[owner_of[winner]]["playoffWins"] += 1
                for name in extra.get(winner, []):
                    bucket(name)["playoffWins"] += 1
            if loser in owner_of:
                out[owner_of[loser]]["playoffLosses"] += 1
                for name in extra.get(loser, []):
                    bucket(name)["playoffLosses"] += 1

        for match in bracket:
            if match.get("p") == 1 and match.get("w") in owner_of:
                champions[season] = owner_of[match["w"]]

    for record in out.values():
        record["playoffYears"] = sorted(record["playoffYears"])
        record["seasons"] = sorted(record["seasons"])
    return out, champions
```

```python
import argparse
import datetime


class SelfCheckError(Exception):
    """Raised when a derived figure disagrees with a curated one."""


def _merge(espn, sleeper, name):
    a = espn.get(name) or {"wins": 0, "losses": 0, "playoffWins": 0,
                           "playoffLosses": 0, "playoffYears": [], "seasons": []}
    b = sleeper.get(name) or {"wins": 0, "losses": 0, "playoffWins": 0,
                              "playoffLosses": 0, "playoffYears": [], "seasons": []}
    return a, b


def build(overrides, espn_raw, chain):
    ids = resolve_identities(overrides)
    espn = espn_era(espn_raw, ids)
    sleeper, champions = sleeper_era(chain, ids, overrides["rosterCredits"])

    lore = {}
    for ring in overrides["rings"]:
        lore[(ring["year"], ring["manager"])] = ring

    ring_years = {}
    for season, name in champions.items():
        ring_years.setdefault(name, set()).add(int(season))
    for ring in overrides["rings"]:
        ring_years.setdefault(ring["manager"], set()).add(ring["year"])

    managers = []
    problems = []

    for name in overrides["managers"]:
        a, b = _merge(espn, sleeper, name)

        wins = a["wins"] + b["wins"]
        losses = a["losses"] + b["losses"]
        p_wins = a["playoffWins"] + b["playoffWins"]
        p_losses = a["playoffLosses"] + b["playoffLosses"]
        years = sorted(set(a["playoffYears"]) | set(b["playoffYears"]))
        tenure = sorted(set(a["seasons"]) | set(b["seasons"]))

        override = overrides["playoffRecordOverrides"].get(name)
        if override:
            p_wins, p_losses = (int(x) for x in override["record"].split("-"))
        else:
            curated = overrides["curatedPlayoffRecords"].get(name)
            if curated and curated != f"{p_wins}-{p_losses}":
                problems.append(
                    f"{name}: derived {p_wins}-{p_losses} but curated {curated}"
                )

        managers.append({
            "name": name,
            "espn": {"wins": a["wins"], "losses": a["losses"]},
            "sleeper": {"wins": b["wins"], "losses": b["losses"]},
            "total": {
                "wins": wins,
                "losses": losses,
                "pct": (wins / (wins + losses)) if wins + losses else 0.0,
            },
            "playoffs": {
                "made": len(years),
                "possible": len(tenure),
                "years": years,
                "record": {"wins": p_wins, "losses": p_losses},
            },
            "rings": [
                {
                    "year": year,
                    "mickeyMouse": bool(lore.get((year, name), {}).get("mickeyMouse")),
                    "note": lore.get((year, name), {}).get("note", ""),
                }
                for year in sorted(ring_years.get(name, set()))
            ],
        })

    if problems:
        raise SelfCheckError(
            "derived figures disagree with curated values:\n  "
            + "\n  ".join(problems)
            + "\nAdd an entry to playoffRecordOverrides to accept the curated value."
        )

    return {
        "generatedAt": datetime.datetime.now(datetime.timezone.utc)
        .replace(microsecond=0)
        .isoformat(),
        "seasons": {
            "espn": [int(y) for y in sorted(espn_raw)],
            "sleeper": sorted(int(c["season"]) for c in chain),
        },
        "managers": sorted(managers, key=lambda m: -m["total"]["wins"]),
    }


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--refresh", action="store_true",
                        help="hit the Sleeper API and update the cache")
    args = parser.parse_args()

    overrides = load_overrides()
    if args.refresh:
        chain = fetch_chain(overrides["leagueId"])
        save_cache(chain)
        print(f"fetched {len(chain)} completed seasons")
    else:
        chain = load_cache()

    espn_raw = {}
    for year, path in ESPN_PATHS.items():
        with open(path, encoding="utf-8") as handle:
            espn_raw[year] = json.load(handle)

    doc = build(overrides, espn_raw, chain)
    with open(OUTPUT_PATH, "w", encoding="utf-8") as handle:
        json.dump(doc, handle, indent=1)
    print(f"wrote {OUTPUT_PATH} for {len(doc['managers'])} managers")


if __name__ == "__main__":
    main()
```

```js
  if (!headerRow) return;

function visibleRows() {

      const va = col.value(a);
      const vb = col.value(b);
      if (typeof va === "number" && typeof vb === "number") {

    if (SORT.key !== key) SORT = { key, direction: "asc" };
    else if (SORT.direction === "asc") SORT = { key, direction: "desc" };
```
